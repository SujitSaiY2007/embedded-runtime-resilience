cmake_minimum_required(VERSION 3.16)
project(embedded_runtime_reference_v2 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  set(REF_WARNING_FLAGS /W4 /WX)
else()
  set(REF_WARNING_FLAGS -Wall -Wextra -Werror -pedantic)
endif()
add_library(ref_runtime_v2 STATIC src/ref_runtime.c)
target_include_directories(ref_runtime_v2 PUBLIC include)
target_compile_options(ref_runtime_v2 PRIVATE ${REF_WARNING_FLAGS})
add_executable(reference_v2_edge_tests tests/test_runtime_edge_cases.c)
target_link_libraries(reference_v2_edge_tests PRIVATE ref_runtime_v2)
target_compile_options(reference_v2_edge_tests PRIVATE ${REF_WARNING_FLAGS})
enable_testing()
add_test(NAME reference_v2_edge_tests COMMAND reference_v2_edge_tests)
